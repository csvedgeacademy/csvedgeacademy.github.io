<script>
const speech = window.speechSynthesis;
let utter;

function playVoice(){
  speech.cancel();

  const accent = document.getElementById("accentSelect").value;
  const texts = document.querySelectorAll(".voice-text");
  let index = 0;

  function speakNext(){
    if(index >= texts.length) return;

    texts.forEach(t=>t.classList.remove("voice-active"));
    texts[index].classList.add("voice-active");

    utter = new SpeechSynthesisUtterance(texts[index].innerText);
    utter.lang = accent;
    utter.rate = 0.9;

    utter.onend = ()=>{
      texts[index].classList.remove("voice-active");
      index++;
      speakNext();
    };

    speech.speak(utter);
  }

  speakNext();
}

function stopVoice(){
  speech.cancel();
  document.querySelectorAll(".voice-text")
    .forEach(t=>t.classList.remove("voice-active"));
}
</script>
